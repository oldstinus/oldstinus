#!/usr/bin/env python3
# -*- coding: utf-8 -*-

"""
Hydraulisch model koker/sas
- GUI file selectie
- Waterinfo parsing
- ADCP koppeling
- Filtering
- Covariantie + cross-correlatie
- Model fit (sqrt(delta h))
- Predictie
"""

import pandas as pd
import numpy as np
import tkinter as tk
from tkinter import filedialog, messagebox
import matplotlib.pyplot as plt
from scipy.signal import correlate
from scipy.stats import median_abs_deviation
from sklearn.linear_model import LinearRegression

# =========================
# CSV PARSER (Waterinfo)
# =========================
def parse_waterinfo(path):
    df = pd.read_csv(path, sep=';', comment='#', encoding='latin1', on_bad_lines='skip')
    
    df.columns = ["time", "value", "qual", "x1", "x2"]
    df["time"] = pd.to_datetime(df["time"], errors='coerce')
    df["value"] = df["value"].astype(str).str.replace(",", ".").astype(float)
    
    return df[["time", "value"]].dropna()

# =========================
# ADCP PARSER
# =========================
def parse_adcp(path):
    df = pd.read_csv(path, sep='\t')
    
    df["time"] = pd.to_datetime(df.iloc[:,0], errors='coerce', dayfirst=True)
    df["v"] = pd.to_numeric(df.iloc[:,4], errors='coerce')
    df["pressure"] = pd.to_numeric(df.iloc[:,11], errors='coerce')
    
    return df[["time", "v", "pressure"]].dropna()

# =========================
# FILTERING
# =========================
def remove_spikes(series, thresh=5):
    mad = median_abs_deviation(series)
    med = np.median(series)
    mask = np.abs(series - med) < thresh * mad
    return series.where(mask)

# =========================
# MAIN ANALYSIS
# =========================
def run_analysis(adcp_file, sea_file, dock_file):
    
    adcp = parse_adcp(adcp_file)
    sea = parse_waterinfo(sea_file)
    dock = parse_waterinfo(dock_file)
    
    # resample
    adcp = adcp.set_index("time").resample("5min").mean()
    sea = sea.set_index("time").resample("5min").mean()
    dock = dock.set_index("time").resample("5min").mean()
    
    # merge
    df = adcp.join(sea.rename(columns={"value":"h_sea"}))
    df = df.join(dock.rename(columns={"value":"h_dock"}))
    
    df = df.interpolate().dropna()
    
    # filtering
    df["v"] = remove_spikes(df["v"])
    df = df.dropna()
    
    # delta h
    df["dh"] = df["h_sea"] - df["h_dock"]
    df["dh_sqrt"] = np.sign(df["dh"]) * np.sqrt(np.abs(df["dh"]))
    
    # =========================
    # COVARIANTIE
    # =========================
    cov = np.cov(df["dh_sqrt"], df["v"])[0,1]
    corr = np.corrcoef(df["dh_sqrt"], df["v"])[0,1]
    
    print("\n=== RESULTATEN ===")
    print("Cov(sqrt(dh), v) =", cov)
    print("Corr =", corr)
    
    # =========================
    # CROSS-CORRELATIE
    # =========================
    x = df["dh_sqrt"].values
    y = df["v"].values
    
    corr_full = correlate(y, x, mode='full')
    lags = np.arange(-len(x)+1, len(x))
    
    best_lag = lags[np.argmax(corr_full)]
    dt = 5 * 60  # 5 min
    
    lag_min = best_lag * dt / 60
    
    print("Lag (min) =", lag_min)
    
    # =========================
    # MODEL FIT
    # =========================
    X = df["dh_sqrt"].values.reshape(-1,1)
    y = df["v"].values
    
    model = LinearRegression()
    model.fit(X, y)
    
    C1 = model.coef_[0]
    C0 = model.intercept_
    
    print("\n=== MODEL ===")
    print(f"v = {C1:.3f} * sqrt(dh) + {C0:.3f}")
    
    # =========================
    # PREDICTIE
    # =========================
    df["v_pred"] = model.predict(X)
    
    # =========================
    # PLOT
    # =========================
    plt.figure(figsize=(10,6))
    plt.plot(df.index, df["v"], label="gemeten")
    plt.plot(df.index, df["v_pred"], label="model")

    plt.title("Model vs meting")
    plt.show()

#!/usr/bin/env python3
# -*- coding: utf-8 -*-

import pandas as pd
import numpy as np
import tkinter as tk
from tkinter import filedialog
import matplotlib.pyplot as plt
from scipy.stats import median_abs_deviation
from sklearn.linear_model import LinearRegression
from sklearn.metrics import mean_squared_error, r2_score

PIPE_AREA = 1.0  # aanpassen

# =========================
# PARSERS
# =========================
def parse_waterinfo(path):
    df = pd.read_csv(path, sep=';', comment='#', encoding='latin1', on_bad_lines='skip')
    df.columns = ["time", "value", "qual", "x1", "x2"]
    df["time"] = pd.to_datetime(df["time"], errors='coerce')
    df["value"] = df["value"].astype(str).str.replace(",", ".").astype(float)
    return df[["time", "value"]].dropna()

def parse_adcp(path):
    df = pd.read_csv(path, sep='\t')
    df["time"] = pd.to_datetime(df.iloc[:,0], errors='coerce', dayfirst=True)
    df["v"] = pd.to_numeric(df.iloc[:,4], errors='coerce')
    return df[["time", "v"]].dropna()

# =========================
# FILTER
# =========================
def remove_spikes(series):
    mad = median_abs_deviation(series)
    med = np.median(series)
    return series.where(np.abs(series - med) < 5 * mad)

# =========================
# HYSTERESE ENERGIE
# =========================
def hysteresis_energy(df):
    dh = df["dh"].values
    Q = df["Q"].values
    
    energy = 0
    for i in range(len(dh)-1):
        d_dh = dh[i+1] - dh[i]
        energy += Q[i] * d_dh
    
    return energy

# =========================
# MAIN
# =========================
def run(adcp_file, sea_file, dock_file):

    adcp = parse_adcp(adcp_file)
    sea = parse_waterinfo(sea_file)
    dock = parse_waterinfo(dock_file)

    # resample
    adcp = adcp.set_index("time").resample("5min").mean()
    sea = sea.set_index("time").resample("5min").mean()
    dock = dock.set_index("time").resample("5min").mean()

    df = adcp.join(sea.rename(columns={"value":"h_sea"}))
    df = df.join(dock.rename(columns={"value":"h_dock"}))
    df = df.interpolate().dropna()

    # filter
    df["v"] = remove_spikes(df["v"])
    df = df.dropna()

    # hydraulica
    df["dh"] = df["h_sea"] - df["h_dock"]
    df["dh_sqrt"] = np.sign(df["dh"]) * np.sqrt(np.abs(df["dh"]))
    df["dh_dt"] = np.gradient(df["dh"], 300)  # 5 min

    df["Q"] = PIPE_AREA * df["v"]

    # =========================
    # MODEL FIT (multi)
    # =========================
    X = np.column_stack([df["dh_sqrt"], df["dh"], df["dh_dt"]])
    y = df["v"].values

    model = LinearRegression()
    model.fit(X, y)

    C1, C2, C3 = model.coef_
    C0 = model.intercept_

    df["v_model"] = model.predict(X)
    df["Q_model"] = PIPE_AREA * df["v_model"]

    print("\n=== MODEL ===")
    print(f"v = {C1:.3f}*sqrt(dh) + {C2:.3f}*dh + {C3:.3f}*d(dh)/dt + {C0:.3f}")

    # =========================
    # FOUTANALYSE
    # =========================
    rmse = np.sqrt(mean_squared_error(y, df["v_model"]))
    r2 = r2_score(y, df["v_model"])
    bias = np.mean(y - df["v_model"])

    print("\n=== FOUT ===")
    print("RMSE:", rmse)
    print("R²:", r2)
    print("Bias:", bias)

    # =========================
    # HYSTERESE
    # =========================
    energy = hysteresis_energy(df)
    print("\n=== ENERGIEVERLIES ===")
    print("Hysterese energie:", energy)

    flood = df[df["dh"] > 0]
    ebb = df[df["dh"] < 0]

    # =========================
    # PLOTS
    # =========================
    plt.figure()
    plt.scatter(df["dh_sqrt"], df["v"], s=5)
    plt.title("v vs sqrt(dh)")

    plt.figure()
    plt.scatter(df["dh"], df["Q"], s=5)
    plt.title("Q vs dh")

    plt.figure()
    plt.scatter(flood["dh"], flood["Q"], s=5, label="vloed")
    plt.scatter(ebb["dh"], ebb["Q"], s=5, label="eb")
    plt.legend()
    plt.title("Hysterese")

    plt.figure()
    plt.plot(df.index, df["Q"], label="gemeten")
    plt.plot(df.index, df["Q_model"], label="model")
    plt.legend()
    plt.title("Debiet")

    plt.figure()
    plt.plot(df.index, y - df["v_model"])
    plt.title("Residuen")

    plt.show()

    return df

    df["v"] = remove_spikes(df["v"])
    df = df.dropna()

    # hydraulica
    df["dh"] = df["h_sea"] - df["h_dock"]
    df["dh_sqrt"] = np.sign(df["dh"]) * np.sqrt(np.abs(df["dh"]))
    df["dh_dt"] = np.gradient(df["dh"], 300)

    df["Q"] = PIPE_AREA * df["v"]

    # =========================
    # MODEL
    # =========================
    X = np.column_stack([df["dh_sqrt"], df["dh"], df["dh_dt"]])
    y = df["v"].values

    model = LinearRegression()
    model.fit(X, y)

    C1, C2, C3 = model.coef_
    C0 = model.intercept_

    df["v_model"] = model.predict(X)
    df["Q_model"] = PIPE_AREA * df["v_model"]

    # =========================
    # FOUT
    # =========================
    rmse = np.sqrt(mean_squared_error(y, df["v_model"]))
    r2 = r2_score(y, df["v_model"])
    bias = np.mean(y - df["v_model"])

    # =========================
    # HYSTERESE ENERGIE
    # =========================
    energy = np.sum(df["Q"].values[:-1] * np.diff(df["dh"].values))

    return df, C1, C2, C3, C0, rmse, r2, bias, energy

from pathlib import Path

g = 9.81

# =========================
# PARSERS
# =========================
def parse_waterinfo(path):
    df = pd.read_csv(path, sep=';', comment='#', encoding='latin1', on_bad_lines='skip')
    df.columns = ["time", "value", "qual", "x1", "x2"]
    df["time"] = pd.to_datetime(df["time"])
    df["value"] = df["value"].astype(str).str.replace(",", ".").astype(float)
    return df[["time","value"]].dropna()

def parse_adcp(path):
    df = pd.read_csv(path, sep='\t')
    df["time"] = pd.to_datetime(df.iloc[:,0], dayfirst=True)
    df["v"] = pd.to_numeric(df.iloc[:,4], errors='coerce')
    return df[["time","v"]].dropna()

# =========================
# FILTERS
# =========================
def lowpass(data, dt, cutoff_hours=6):
    fs = 1/dt
    fc = 1/(cutoff_hours*3600)
    b,a = butter(3, fc/(fs/2), btype='low')
    return filtfilt(b,a,data)

def highpass(data, dt, cutoff_hours=2):
    fs = 1/dt
    fc = 1/(cutoff_hours*3600)
    b,a = butter(3, fc/(fs/2), btype='high')
    return filtfilt(b,a,data)

# =========================
# HYSTERESE PER CYCLUS
# =========================
def compute_cycles(df):
    dh = df["dh"].values
    zero_cross = np.where(np.diff(np.sign(dh)))[0]
    
    energies = []
    for i in range(len(zero_cross)-1):
        seg = df.iloc[zero_cross[i]:zero_cross[i+1]]
        E = np.sum(seg["Q"].values[:-1] * np.diff(seg["dh"].values))
        energies.append(E)
    return energies

# =========================
# FFT analyse
# =========================
def fft_analysis(signal, dt):
    N = len(signal)
    yf = fft(signal)
    xf = fftfreq(N, dt)
    return xf, np.abs(yf)

# =========================
# MAIN
# =========================
def run(adcp_file, sea_file, dock_file):

    adcp = parse_adcp(adcp_file)
    sea = parse_waterinfo(sea_file)
    dock = parse_waterinfo(dock_file)

    # resample
    adcp = adcp.set_index("time").resample("5min").mean()
    sea = sea.set_index("time").resample("5min").mean()
    dock = dock.set_index("time").resample("5min").mean()

    df = adcp.join(sea.rename(columns={"value":"h_sea"}))
    df = df.join(dock.rename(columns={"value":"h_dock"}))
    df = df.interpolate().dropna()

    dt = 300

    df["dh"] = df["h_sea"] - df["h_dock"]
    df["dh_sqrt"] = np.sign(df["dh"]) * np.sqrt(np.abs(df["dh"]))

    # =========================
    # KALIBRATIE
    # =========================
    X = df["dh_sqrt"].values.reshape(-1,1)
    y = df["v"].values

    model = HuberRegressor().fit(X,y)
    C = model.coef_[0]

    df["v_model"] = model.predict(X)

    print("\n=== KALIBRATIE ===")
    print("C =", C)

    # =========================
    # DEBIET
    # =========================
    A_eff = 1.0  # schaalfactor
    df["Q"] = A_eff * df["v"]

    # =========================
    # HYSTERESE
    # =========================
    energies = compute_cycles(df)
    print("Gemiddelde energie:", np.mean(energies))

    # =========================
    # POMPDETECTIE
    # =========================
    v_hp = highpass(df["v"].values, dt)

    xf, yf = fft_analysis(v_hp, dt)

    # =========================
    # PLOTS

    plt.title("Debietcurve")

    plt.figure()
    plt.plot(xf, yf)

    plt.figure()
    plt.hist(v_hp, bins=100)
    plt.title("Pomp signaal")

    plt.show()

    # =========================
    # RAPPORT
    # =========================
    html = f"""
    <html><body>
    <h1>WL Hydraulisch Rapport</h1>

    <h2>Kalibratie</h2>
    C = {C:.3f}

    <h2>Hysterese</h2>
    Gemiddelde energie = {np.mean(energies):.3f}

    <h2>Interpretatie</h2>
    <ul>
    <li>Dominante stroming: getij</li>
    <li>Pompactiviteit: aanwezig in FFT</li>
    <li>Niet-lineair gedrag bevestigd</li>
    </ul>

    </body></html>

    Path("WL_rapport.html").write_text(html)

#!/usr/bin/env python3
# -*- coding: utf-8 -*-

import pandas as pd
import numpy as np
import tkinter as tk
from tkinter import filedialog
import matplotlib.pyplot as plt
from sklearn.linear_model import LinearRegression

# =========================
# SETTINGS
# =========================
DT = 300  # 5 min
WINDOW = 20  # smoothing

# =========================
# PARSERS
# =========================
def parse_waterinfo(path):
    df = pd.read_csv(path, sep=';', comment='#', encoding='latin1', on_bad_lines='skip')
    df.columns = ["time", "value", "qual", "x1", "x2"]
    df["time"] = pd.to_datetime(df["time"])
    df["value"] = df["value"].astype(str).str.replace(",", ".").astype(float)
    return df[["time","value"]].dropna()

def parse_adcp(path):
    df = pd.read_csv(path, sep='\t')
    df["time"] = pd.to_datetime(df.iloc[:,0], dayfirst=True)
    df["v"] = pd.to_numeric(df.iloc[:,4], errors='coerce')
    return df[["time","v"]].dropna()

# =========================
# TRAIN MODEL
# =========================
def train_model(df):

    df["dh"] = df["h_sea"] - df["h_dock"]
    df["dh_sqrt"] = np.sign(df["dh"]) * np.sqrt(np.abs(df["dh"]))
    df["dh_dt"] = np.gradient(df["dh"], DT)
    df["v_lag"] = df["v"].shift(1)

    df = df.dropna()

    X = np.column_stack([
        df["dh_sqrt"],
        df["dh"],
        df["dh_dt"],
        df["v_lag"]
    ])

    y = df["v"]

    model = LinearRegression().fit(X, y)
    return model

# =========================
# REALTIME PREDICT
# =========================
def predict_stream(df, model):

    preds = []
    v_prev = df["v"].iloc[0]

    for i in range(len(df)):

        dh = df["h_sea"].iloc[i] - df["h_dock"].iloc[i]
        dh_sqrt = np.sign(dh)*np.sqrt(abs(dh))
        dh_dt = 0 if i==0 else (df["dh"].iloc[i] - df["dh"].iloc[i-1]) / DT

        X = np.array([[dh_sqrt, dh, dh_dt, v_prev]])

        v_pred = model.predict(X)[0]
        preds.append(v_pred)

        v_prev = v_pred

    df["v_pred"] = preds
    return df

# =========================
# LIMIT DETECTION
# =========================
def detect_limits(df):

    # methode 1: helling
    df["slope"] = np.gradient(df["v"], df["dh_sqrt"])

    # methode 2: max debiet
    vmax = np.percentile(df["v"], 95)

    df["limit_flag"] = (df["v"] > vmax*0.95) | (df["slope"] < 0.1)

    print("\n=== LIMIET DETECTIE ===")
    print("Max snelheid:", vmax)

    return df

# =========================
# MAIN
# =========================
def run(adcp_file, sea_file, dock_file):

    adcp = parse_adcp(adcp_file)
    sea = parse_waterinfo(sea_file)
    dock = parse_waterinfo(dock_file)

    # resample
    adcp = adcp.set_index("time").resample("5min").mean()
    sea = sea.set_index("time").resample("5min").mean()
    dock = dock.set_index("time").resample("5min").mean()

    df = adcp.join(sea.rename(columns={"value":"h_sea"}))
    df = df.join(dock.rename(columns={"value":"h_dock"}))
    df = df.interpolate().dropna()

    # train
    model = train_model(df)

    # predict
    df = predict_stream(df, model)

    # limits
    df = detect_limits(df)

    # =========================
    # PLOTS
    # =========================
    plt.figure(figsize=(10,5))
    plt.plot(df.index, df["v"], label="gemeten")
    plt.plot(df.index, df["v_pred"], label="voorspeld")
    plt.legend()
    plt.title("Realtime voorspelling")

    plt.figure()
    plt.scatter(df["dh"], df["v"], s=5)
    plt.scatter(df[df["limit_flag"]]["dh"], df[df["limit_flag"]]["v"], color="red")
    plt.title("Debietlimiet detectie")

    plt.show()

    return df

# =========================
# GUI
# =========================
def main():

    root = tk.Tk()
    root.withdraw()

    adcp = filedialog.askopenfilename(title="ADCP")
    sea = filedialog.askopenfilename(title="Zee")
    dock = filedialog.askopenfilename(title="Dok")

    run(adcp, sea, dock)

if __name__ == "__main__":
    main()

#!/usr/bin/env python3
# -*- coding: utf-8 -*-

import pandas as pd
import numpy as np
import tkinter as tk
from tkinter import filedialog
import matplotlib.pyplot as plt
import zipfile, glob, os

# =========================
# WATERINFO PARSER
# =========================
def load_waterinfo_from_zip(zip_path):

    tmp_dir = "/tmp_extract_" + os.path.basename(zip_path).replace(" ", "_")
    os.makedirs(tmp_dir, exist_ok=True)

    with zipfile.ZipFile(zip_path, 'r') as z:
        z.extractall(tmp_dir)

    csvs = [c for c in glob.glob(tmp_dir + "/*.csv") if "disclaimer" not in c.lower()]

    df = pd.read_csv(csvs[0], sep=';', comment='#', encoding='latin1', on_bad_lines='skip')
    df.columns = ["time","value","q","x1","x2"]

    df["time"] = pd.to_datetime(df["time"], errors='coerce').dt.tz_localize(None)
    df["value"] = df["value"].astype(str).str.replace(",",".").astype(float)

    return df[["time","value"]].dropna()

# =========================
# ADCP PARSER
# =========================
def load_adcp(path):

    df = pd.read_csv(path, sep="\t")

    df["time"] = pd.to_datetime(df.iloc[:,0], errors='coerce', dayfirst=True).dt.tz_localize(None)
    df["v"] = pd.to_numeric(df.iloc[:,4], errors='coerce')

    return df[["time","v"]].dropna()

# =========================
# MAIN
# =========================
def run(adcp_file, sea_zip, dock_zip):

    # load
    adcp = load_adcp(adcp_file)
    sea = load_waterinfo_from_zip(sea_zip)
    dock = load_waterinfo_from_zip(dock_zip)

    # resample
    adcp = adcp.set_index("time").resample("5min").mean()
    sea = sea.set_index("time").resample("5min").mean()
    dock = dock.set_index("time").resample("5min").mean()

    # =========================
    # DEBUG (BELANGRIJK)
    # =========================
    print("\n=== TIJDBEREIKEN ===")
    print("ADCP :", adcp.index.min(), "→", adcp.index.max())
    print("SEA  :", sea.index.min(), "→", sea.index.max())
    print("DOCK :", dock.index.min(), "→", dock.index.max())

    # =========================
    # OVERLAP AUTOMATISCH
    # =========================
    start = max(adcp.index.min(), sea.index.min(), dock.index.min())
    end   = min(adcp.index.max(), sea.index.max(), dock.index.max())

    print("\n=== OVERLAP ===")
    print("START:", start)
    print("END  :", end)

    # check
    if start >= end:
        print("\n❌ GEEN OVERLAP → controleer data")
        return

    # slice
    adcp = adcp.loc[start:end]
    sea = sea.loc[start:end]
    dock = dock.loc[start:end]

    # merge
    df = adcp.join(sea.rename(columns={"value":"h_sea"}))
    df = df.join(dock.rename(columns={"value":"h_dock"}))

    df = df.interpolate().dropna()

    print("\nAantal punten:", len(df))

    # =========================
    # BEREKENINGEN
    # =========================
    df["dh"] = df["h_sea"] - df["h_dock"]
    df["dh_sqrt"] = np.sign(df["dh"]) * np.sqrt(np.abs(df["dh"]))

    # regressie
    coef = np.polyfit(df["dh_sqrt"], df["v"], 1)

    print("\n=== RELATIE ===")
    print(f"v = {coef[0]:.3f} * sqrt(dh) + {coef[1]:.3f}")

    # voorspelling
    df["v_model"] = np.polyval(coef, df["dh_sqrt"])

    # metrics
    res = df["v"] - df["v_model"]
    rmse = np.sqrt(np.mean(res**2))
    r2 = 1 - np.sum(res**2)/np.sum((df["v"]-df["v"].mean())**2)

    print("\n=== KWALITEIT ===")
    print("RMSE:", round(rmse,3))
    print("R²  :", round(r2,3))

    # hysterese energie
    energy = np.sum(df["v"].values[:-1] * np.diff(df["dh"].values))
    print("\nHysterese energie:", round(energy,3))

    # limiet
    vmax = np.percentile(df["v"], 95)
    print("v_max (95%):", round(vmax,3))

    # =========================
    # GRAFIEKEN
    # =========================

    # v vs sqrt(dh)
    plt.figure()
    plt.scatter(df["dh_sqrt"], df["v"], s=5)
    plt.plot(df["dh_sqrt"], df["v_model"], color="red")
    plt.title("v vs sqrt(dh)")
    plt.xlabel("sqrt(dh)")
    plt.ylabel("v")
    plt.grid()

    # hysterese
    plt.figure()
    plt.scatter(df["dh"], df["v"], s=5)
    plt.title("Hysterese")
    plt.xlabel("dh")
    plt.ylabel("v")
    plt.grid()

    # tijdreeks
    plt.figure(figsize=(10,5))
    plt.plot(df.index, df["v"], label="gemeten")
    plt.plot(df.index, df["v_model"], label="model")
    plt.legend()
    plt.title("Tijdreeks")

    plt.show()

# =========================
# GUI
# =========================
def main():

    root = tk.Tk()
    root.withdraw()

    adcp = filedialog.askopenfilename(title="ADCP TSV")
    sea = filedialog.askopenfilename(title="Zeepeil ZIP")
    dock = filedialog.askopenfilename(title="Dokpeil ZIP")

    run(adcp, sea, dock)

if __name__ == "__main__":
    main()

#!/usr/bin/env python3
# -*- coding: utf-8 -*-

import pandas as pd
import numpy as np
import tkinter as tk
from tkinter import filedialog, simpledialog
import matplotlib.pyplot as plt
import zipfile, glob, os
from pathlib import Path

OUTDIR = "output_report"
os.makedirs(OUTDIR, exist_ok=True)

# =========================
# LOADERS
# =========================
def load_waterinfo(zip_path):

    tmp = OUTDIR + "/tmp_" + os.path.basename(zip_path).replace(" ", "_")
    os.makedirs(tmp, exist_ok=True)

    with zipfile.ZipFile(zip_path, 'r') as z:
        z.extractall(tmp)

    csvs = [c for c in glob.glob(tmp + "/*.csv") if "disclaimer" not in c.lower()]
    df = pd.read_csv(csvs[0], sep=';', comment='#', encoding='latin1')

    df.columns = ["time","value","q","x1","x2"]
    df["time"] = pd.to_datetime(df["time"]).dt.tz_localize(None)
    df["value"] = df["value"].astype(str).str.replace(",",".").astype(float)

    return df[["time","value"]].dropna()

def load_adcp(path):

    df = pd.read_csv(path, sep="\t")
    df["time"] = pd.to_datetime(df.iloc[:,0], dayfirst=True).dt.tz_localize(None)
    df["v"] = pd.to_numeric(df.iloc[:,4], errors='coerce')

    return df[["time","v"]].dropna()

# =========================
# MAIN
# =========================
def run(adcp_file, sea_zip, dock_zip):

    # vraag doorsnede
    root = tk.Tk()
    root.withdraw()
    A = simpledialog.askfloat("Doorsnede", "Geef doorsnede (m²):", initialvalue=1.0)

    adcp = load_adcp(adcp_file)
    sea = load_waterinfo(sea_zip)
    dock = load_waterinfo(dock_zip)

    # resample
    adcp = adcp.set_index("time").resample("5min").mean()
    sea = sea.set_index("time").resample("5min").mean()
    dock = dock.set_index("time").resample("5min").mean()

    # overlap
    start = max(adcp.index.min(), sea.index.min(), dock.index.min())
    end = min(adcp.index.max(), sea.index.max(), dock.index.max())

    adcp = adcp.loc[start:end]
    sea = sea.loc[start:end]
    dock = dock.loc[start:end]

    df = adcp.join(sea.rename(columns={"value":"h_sea"}))
    df = df.join(dock.rename(columns={"value":"h_dock"}))
    df = df.interpolate().dropna()

    # hydraulica
    df["dh"] = df["h_sea"] - df["h_dock"]
    df["dh_abs"] = np.sqrt(np.abs(df["dh"]))

    # debiet
    df["Q"] = A * df["v"]

    # regimes
    flood = df[df["dh"] > 0]
    ebb = df[df["dh"] < 0]

    # regressie
    Cf = np.polyfit(flood["dh_abs"], flood["Q"], 1)
    Ce = np.polyfit(ebb["dh_abs"], ebb["Q"], 1)

    print("\n=== DEBIETMODEL ===")
    print(f"Flood: Q = {Cf[0]:.3f} * sqrt(dh)")
    print(f"Ebb  : Q = {Ce[0]:.3f} * sqrt(|dh|)")

    # =========================
    # FIGUREN
    # =========================

    # debietcurve
    plt.figure()
    plt.scatter(df["dh"], df["Q"], s=5)

    x = np.linspace(df["dh"].min(), df["dh"].max(), 200)
    plt.plot(x[x>0], Cf[0]*np.sqrt(x[x>0]), label="Flood model")
    plt.plot(x[x<0], Ce[0]*np.sqrt(-x[x<0]), label="Ebb model")

    plt.legend()
    plt.title("Debietcurve (m³/s)")
    plt.xlabel("Δh")
    plt.ylabel("Q")

    f1 = OUTDIR + "/debietcurve.png"
    plt.savefig(f1)
    plt.close()

    # tijdreeks
    plt.figure(figsize=(10,5))
    plt.plot(df.index, df["Q"])
    plt.title("Debiet tijdreeks")
    f2 = OUTDIR + "/tijdreeks_Q.png"
    plt.savefig(f2)
    plt.close()

    # =========================
    # RAPPORT
    # =========================
    html = f"""
    <html><body>

    <h1>WL Debietrapport</h1>

    <h2>Doorsnede</h2>
    A = {A} m²

    <h2>Model</h2>
    Flood: Q = {Cf[0]:.3f} √Δh<br>
    Ebb: Q = {Ce[0]:.3f} √|Δh|

    <h2>Interpretatie</h2>
    <ul>
    <li>Niet-lineair gedrag bevestigd</li>
    <li>Verschil tussen ebb en flood = hysterese</li>
    <li>Debiet gelimiteerd door hydraulische capaciteit</li>
    </ul>

    <h2>Debietcurve</h2>
    <img src="debietcurve.png" width="700">

    <h2>Tijdreeks</h2>
    <img src="tijdreeks_Q.png" width="900">

    </body></html>
    """

    Path(OUTDIR + "/debietrapport.html").write_text(html)

    print("\n✅ Debietrapport klaar in:", OUTDIR)

# =========================
# GUI
# =========================
def main():

    root = tk.Tk()
    root.withdraw()

    adcp = filedialog.askopenfilename(title="ADCP TSV")
    sea = filedialog.askopenfilename(title="Zee ZIP")
    dock = filedialog.askopenfilename(title="Dok ZIP")

    run(adcp, sea, dock)

if __name__ == "__main__":
    main()
