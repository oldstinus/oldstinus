  Dim verzend_ftp_data_2
  Dim verzend_ftp_data_3
  Dim verzend_ftp_data_4
  Dim RemoteFileName_1 As String * 100
  Dim RemoteFileName_2 As String * 100
  Dim RemoteFileName_3 As String * 100
  Dim RemoteFileName_4 As String * 100
  Dim Tijd_code As String * 25
  Dim Tijd_string As String * 200
  Dim Tijd_header As String * 200
  Dim Huidige_tijd As String * 100
  Dim delta_tijd
  Dim Tijd_array(7)
  Dim Tijd_dev_start
  Dim Tijd_dev_set
  Dim Schrikkel_jaar
  Dim Ontvangen_tijd_juist

  Scan (Scan_time_slow,Scan_unit_slow,3,0)

    If C_geen_IP >= 60 Then
      Info = 7
      CallTable Info
      Restart
    EndIf

    ' ### Test of modem aangezet moet worden. ###
    ' ### Test of het modem aangezet moet worden op tijd. ###
    Test_1 = (MinInDeDag - Modem_aan_tijd - FIX(MinInDeDag/Modem_aan_uit_int) * Modem_aan_uit_int) MOD Modem_aan_uit_int < Modem_uit_tijd - Modem_aan_tijd
    Test_2 = (MinInDeDag - Modem_aan_tijd - FIX(MinInDeDag/Modem_aan_uit_int) * Modem_aan_uit_int) MOD Modem_aan_uit_int >= 0

    ' ### controle of GPRS aan moet blijven na communicatie ###
    If Verzend_ftp_data = 0 AND Verzend_call_back_data = 0 AND Modem_handmatig_aan = 0 Then
      If Modem_aan_tijd < Modem_uit_tijd Then
        If Test_1 AND Test_2 Then
          If Modem_is_aan = 0 Then Modem_is_aan = 1
        Else
          If Modem_is_aan <> 0 Then
            Sluit_GPRS = 1
            Modem_is_aan = 0
          EndIf
        EndIf

      ElseIf Modem_aan_tijd > Modem_uit_tijd Then
        If Test_1 OR Test_2 Then
          If Modem_is_aan = 0 Then Modem_is_aan = 1
        Else
          If Modem_is_aan <> 0 Then
            Sluit_GPRS = 1
            Modem_is_aan = 0
          EndIf
        EndIf

      ElseIf Modem_is_aan <> 0 Then
        Sluit_GPRS = 1
        Modem_is_aan = 0
      EndIf
    EndIf

    If (Verzend_ftp_data = 1 OR Verzend_call_back_data = 1 OR Modem_handmatig_aan = 1) AND Modem_is_aan = 0 Then Modem_is_aan = 1

    ' ### GPRS/GSM afmelden ###
    If Sluit_GPRS = 1 Then

      PortGet(Modem_status,Modem_cont)        'test of modem nog aan staat, dan uitzetten

      If Modem_status = 1 Then

        PPP_afgesloten = 0
        Modem_info = "PPP afsluiten."

        Timer(0,uSec,2)
        Do
          PPP_afgesloten = PPPClose
          T_PPP_sluiten = Timer(0,uSec,4)/1000000
        Loop Until (PPP_afgesloten = -1) OR (T_PPP_sluiten >= 10)
        Timer(0,uSec,3)

        Modem_info = "Modem afmelden."

        SerialOpen(Modem_port,Modem_baud_rate,0,0,1000)

        Timer(0,uSec,2)
        Do
          modem_afmelden = SerialOut(Modem_port,"at^SMSO"+ CHR(13),"^SHUTDOWN",1,1000)
          T_modem_afmelden = Timer(0,uSec,4)/1000000
        Loop Until (modem_afmelden <> 0) OR (T_modem_afmelden >= 15)
        Timer(0,uSec,3)

        SerialClose(Modem_port)

        Modem_info = "Modem uitzetten."
      EndIf

      Delay(0,1,sec)
      Sluit_GPRS = 0
      PPP_is_open = 0
      PortSet(Modem_cont,0)
      Modem_is_aan = 0
      Modem_info = "Communicatie afgesloten, modem staat uit."
    EndIf

    ' ### GSM/GPRS aanmelden ###
    If Modem_is_aan = 1 Then
      PPP_is_open = 0
      Sluit_GPRS = 1
      T_PPP_sluiten = 0
      T_modem_afmelden = 0
      Modem_info = "Modem aanzetten."

      Do While PPP_is_open = 0

        PortSet (Modem_cont ,1 )                'modem aan zetten
        Delay(0,1,sec)

        SerialOpen(Modem_port,Modem_baud_rate,0,0,1000)

        Timer(0,uSec,2)
        Do
          modem_restart = SerialOut(Modem_port,"A","A",1,50)
          If modem_restart = 0 Then
            modem_restart = SerialOut(Modem_port,"at+ipr=" + Modem_baud_rate + CHR(13),"OK",1,50)
            modem_restart = SerialOut(Modem_port,"atX0" + CHR(13),"OK",1,50)
            modem_restart = SerialOut(Modem_port,"at&d0" + CHR(13),"OK",1,50)
            modem_restart = SerialOut(Modem_port,"at&W" + CHR(13),"OK",1,50)
          EndIf
          T_modem_restart = Timer(0,uSec,4)/1000000
        Loop Until (modem_restart <> 0) OR (T_modem_restart >= 15)
        Timer(0,uSec,3)

        If T_modem_restart >= 15 Then ExitDo

        Delay(0,1,sec)

        GSM_comm_aanmelden = 0
        Modem_info = "Modem aanmelden."

        Timer(0,uSec,2)                'test of modem is aangemeld dan verder anders exit scan
        Do
          GSM_comm_aanmelden = SerialOut(Modem_port,"at+creg?"+ CHR(13),"0,5",1,50)
          If GSM_comm_aanmelden = 0 Then
            GSM_comm_aanmelden = SerialOut(Modem_port,"at+creg?"+ CHR(13),"0,1",1,50)
          EndIf
          T_GSM_comm_aanmelden = Timer(0,uSec,4)/1000000
        Loop Until (GSM_comm_aanmelden <> 0) OR (T_GSM_comm_aanmelden >= 30)
        Timer(0,uSec,3)

        If T_GSM_comm_aanmelden >= 30 Then ExitDo

        '##### CSQ waarde opvragen nog uitwerken.
        SerialFlush (Modem_port)
        SerialOut(Modem_port,"at+csq"+CHR(13),"",0,10)
        SerialIn (CSQ_waarde,Modem_port,100,"OK",25)

        SerialClose(Modem_port)

        Modem_info = "Ip-nummer opvragen."
        PPP_verbinding = "0.0.0.0"

        Timer(0,uSec,2)                'PPP openen. gelukt dan verder ander exit scan
        Do
          PPP_verbinding = PPPOpen
          T_IP_ontvangen = Timer(0,uSec,4)/1000000
        Loop Until (PPP_verbinding <> "0.0.0.0") OR (T_IP_ontvangen >= 40)
        Timer(0,uSec,3)

        If T_IP_ontvangen >= 40 Then ExitDo

        Delay(0,1,Sec)
        PPP_is_open = -1                  'open PPP gelukt ppp_is_open = -1 anders bij volgende scan weer op nieuw
        Modem_is_aan = -1
        Sluit_GPRS = 0
        C_geen_IP = 0
        Modem_info = "Ip-nummer: " + PPP_verbinding + " ontvangen"
      Loop
    EndIf

    ' ### tijdsynchroniseren ###
    'opvragen de tijd bij de website
    If Modem_is_aan = -1 AND Tijd_synchroniseren = 1 Then

      Tijd_synchroniseren = 0
      Timer (6,Sec,2)

      If NTP_server = "" Then

        Tijd_header = ""
        Tijd_synchronistatie_result = HTTPGet (Tijd_URL,Tijd_string,Tijd_header,500)
      Else

        Time_is_Set = 0
        Time_is_Set = NetworkTimeProtocol (NTP_server,Tijd_zone*3600,5000)
      EndIf
    EndIf

    If Time_is_Set <> 0 Then
      Timer (6,Sec,1)
    EndIf

    'na het ontvangen van informatie controle op juiste waarde en tijd goed zetten.
    If NTP_server = "" Then
      If Tijd_synchronistatie_result >= 100 AND Tijd_synchroniseren = -1 Then

        SplitStr (Tijd_array(),Tijd_string,CHR(45)&CHR(58)&CHR(84),6,7)
        Tijd_array(7) = 0

        Ontvangen_tijd_juist = 0

        If Tijd_array(1) >= 2017 AND Tijd_array(1) <= 2050 Then
          If Tijd_array(2) >= 1 AND Tijd_array(2) <= 12 Then
            If Tijd_array(3) >= 1 AND Tijd_array(3) <= 31 Then
              If Tijd_array(4) >= 0 AND Tijd_array(4) <= 23 Then
                If Tijd_array(5) >= 0 AND Tijd_array(5) <= 59 Then
                  If Tijd_array(6) >= 0 AND Tijd_array(6) <= 59 Then
                    Ontvangen_tijd_juist = 1
                  EndIf
                EndIf
              EndIf
            EndIf
          EndIf
        EndIf

        'tijd vweschil tussen start instructie en hier opvragen om de zien of het in de huidige scan is uitgevoerd.
        delta_tijd = Timer (6,mSec,1)

        'huidige tijd van de logger opvragen.
        Huidige_tijd = status.timestamp(1,1)

        'klok variatie nog zonder aanpassing opvragen.
        Tijd_dev_start = ClockChange

        'alleen de tijd zetten als alle goed is ontvangen en binnen de gestelde tijd.
        If delta_tijd < 5000 AND Ontvangen_tijd_juist = 1 Then

          'helft van het tijdverschil tussen het vragen en ontvangen van de tijd van de website er bij tellen
          Tijd_array(7) = FRAC(delta_tijd / 2000) * 1000
          Tijd_array(6) = Tijd_array(6) + INT(delta_tijd / 2000)

          'van UTC tijd naar locale tijd.
          Tijd_array(4) = Tijd_array(4) + Tijd_zone

          'test of we over de 60 seconden gaan.
          'als dat zo is wordt de secondewaarde goed gezet en 1 bij de minuutwaarde geteld.
          If Tijd_array(6) > 59 Then
            Tijd_array(6) = Tijd_array(6) - 60
            Tijd_array (5) = Tijd_array(5) + 1
          EndIf

          'test of we over de 60 minuten gaan.
          'als dat zo is wordt de minuutwaarde goed gezet en 1 bij de uurwaarde geteld.
          If Tijd_array(5) > 59 Then
            Tijd_array(5) = Tijd_array(5) - 60
            Tijd_array (4) = Tijd_array(4) + 1
          EndIf

          'test of we over de 24 uur gaan.
          'als dat zo is wordt de uurwaarde goed gezet en 1 bij de dagwaarde geteld.
          If Tijd_array(4) > 23 Then
            Tijd_array(4) = Tijd_array(4) - 24
            Tijd_array (3) = Tijd_array(3) + 1
          EndIf

          'tes of het een schrikkeljaar is.
          If Tijd_array(1) MOD 4 <> 0 Then
            Schrikkel_jaar = 0
          ElseIf Tijd_array(1) MOD 400 = 0 Then
            Schrikkel_jaar = 1
          ElseIf Tijd_array(1) MOD 100 = 0 Then
            Schrikkel_jaar = 0
          Else
            Schrikkel_jaar = 1
          EndIf

          'test of we te veel dagen in de maand krijgen.
          'maand 2 afhankelijk van het schrikkeljaar 28 of 29.
          'maand 4, 6, 9 en 11 max 30.
          'maand 1, 3, 5, 7, 8, 10 en 12 max 31.
          'bij te veel dag aanpassen en maand 1 ophogen.
          If Tijd_array(3) > 28 AND Tijd_array(2) = 2 AND Schrikkel_jaar = 0 Then
            Tijd_array(3) = Tijd_array(3) - 28
            Tijd_array(2) = Tijd_array(2) + 1
          ElseIf Tijd_array(3) > 29 AND Tijd_array(2) = 2 Then
            Tijd_array(3) = Tijd_array(3) - 29
            Tijd_array(2) = Tijd_array(2) + 1
          ElseIf Tijd_array(3) > 30 AND (Tijd_array(2) = 4 OR Tijd_array(2) = 6 OR Tijd_array(2) = 9 OR Tijd_array(2) = 11) Then
            Tijd_array(3) = Tijd_array(3) - 30
            Tijd_array(2) = Tijd_array(2) + 1
          ElseIf Tijd_array(3) > 31 Then
            Tijd_array(3) = Tijd_array(3) - 31
            Tijd_array(2) = Tijd_array(2) + 1
          EndIf

          'test of we te veel maanden in het jaar krijgen.
          'anders maand aanpassen en het jaar 1 ophogen.
          If Tijd_array(2) > 12 Then
            Tijd_array(2) = Tijd_array(2) - 12
            Tijd_array(1) = Tijd_array(1) + 1
          EndIf

          Tijd_synchroniseren = 0
          ClockSet (Tijd_array())
          Delay (1,1000,mSec)
          Tijd_dev_set = ClockChange

          Info = 1
          CallTable Info
          Tijd_dev_set = 0

        Else
          Tijd_synchroniseren = 0
          Info = 2
          CallTable Info

        EndIf
      EndIf
    EndIf

    ' ### Call Back data verzenden ###
    If Modem_is_aan = -1 AND Verzend_call_back_data = 1 Then

      Verzend_call_back_data = 2
      Timer (5,Sec,2)
      Call_back_Socket = TCPOpen (IP_URL_call_back_server,Call_back_port,0)
      SendVariables (Verzend_call_back_data,Call_back_Socket,4094,4094,0000,1000,"Public","Callback",Info,1)

    EndIf

    If Verzend_call_back_data = 0 Then
      Timer (5,Sec,1)
    EndIf

    ' ### FTP data verzenden ###
    If Modem_is_aan = -1 AND Verzend_ftp_data = 1 Then

      Verzend_ftp_data = 2
      Timer (4,Sec,2)

      verzend_ftp_data_1 = 1
      verzend_ftp_data_2 = 1
      verzend_ftp_data_3 = 1
      verzend_ftp_data_4 = 1

      Tijd_code =  "_" + rTime(1) + "-" + FormatFloat (rTime(2),"%02.0f") + "-" + FormatFloat (rTime(3),"%02.0f") + "_" + FormatFloat (rTime(4),"%02.0f") + FormatFloat (rTime(5),"%02.0f") + FormatFloat (rTime(6),"%02.0f")

      RemoteFileName_1 = "aquatroll/" + Locatie_code + AT_sensor_1_code + Tijd_code + ".dat"
      verzend_ftp_data_1 = FTPClient (IP_URL_ftp_server,User_ftp,Pass_ftp,"MW_AT_1",RemoteFileName_1,2,0,0,Min,1012)

      RemoteFileName_2 = "aquatroll/" + Locatie_code + AT_sensor_2_code + Tijd_code + ".dat"

      #If AT_2_adres <> 0 Then
        verzend_ftp_data_2 = "aquatroll/" + FTPClient (IP_URL_ftp_server,User_ftp,Pass_ftp,"MW_AT_2",RemoteFileName_2,2,0,0,Min,1012)
      #EndIf

      RemoteFileName_3 = "aquatroll/" + Locatie_code + AT_sensor_3_code + Tijd_code + ".dat"
      #If AT_3_adres <> 0 Then
        verzend_ftp_data_3 = FTPClient (IP_URL_ftp_server,User_ftp,Pass_ftp,"MW_AT_3",RemoteFileName_3,2,0,0,Min,1012)
      #EndIf

      RemoteFileName_4 = Locatie_code + AT_sensor_4_code + Tijd_code + ".dat"
      #If AT_4_adres <> 0 Then
        verzend_ftp_data_4 = FTPClient (IP_URL_ftp_server,User_ftp,Pass_ftp,"MW_AT_4",RemoteFileName_4,2,0,0,Min,1012)
      #EndIf

    EndIf

    If verzend_ftp_data_1 < 1 AND verzend_ftp_data_2 < 1 AND verzend_ftp_data_3 < 1 AND verzend_ftp_data_4 < 1 Then
      Verzend_ftp_data = 0
      Timer (4,Sec,1)
    EndIf

    ' ### geen communicatie afvangen ###
    Data_FTP_timer = Timer (4,Sec,4)
    Data_call_back_timer = Timer (5,Sec,4)
    Tijd_synchroniseren_timer = Timer (6,sec,4)

    If Data_FTP_timer >= 300 Then
      Verzend_ftp_data = 0
      Timer (4,Sec,3)
      Info = 4
      CallTable Info
      Sluit_GPRS = 1
      Modem_is_aan = 0
    EndIf

    If Data_call_back_timer >= 300 OR Verzend_call_back_data >= 5 Then
      Verzend_call_back_data = 0
      Timer (5,Sec,3)
      Info = 5
      CallTable Info
      Sluit_GPRS = 1
      Modem_is_aan = 0
      Restart
    EndIf

    If Tijd_synchroniseren_timer >= 90 OR Time_is_Set = NAN Then
      Tijd_synchroniseren = 0
      Timer (6,Sec,3)
      Info = 6
      CallTable Info
      Sluit_GPRS = 1
      Modem_is_aan = 0
    EndIf

    PPP_verbinding_aktief = ComPortIsActive (Call_back_Socket)

  NextScan

EndProg
